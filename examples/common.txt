# Minimal example — copy it elsewhere and adapt it.
#
# A real rig's configuration does NOT live in this repository: it describes CC
# numbers, track indices and ports specific to one installation. Point the bridge
# at yours with the OSCMIDI_CONFIG environment variable.

group   example
port    Ableton Loopback
channel 16
target  127.0.0.1:11000 -> 11001

# ── commands: incoming MIDI → OSC ───────────────────────────────────────────
send  cc 20   /live/song/start_playing
send  cc 21   /live/song/stop_playing
send  cc 24   /live/song/tap_tempo
send  cc 25   $v+50   /live/song/set/tempo $a.0     # 0-127 → 50-177 bpm
send  cc 30   /live/view/set/selected_scene $a
send  cc 31   /live/scene/fire_selected

# ── gestures: computed from the observed state ──────────────────────────────
verb  cc 60   scene.prev
verb  cc 61   scene.next
verb  cc 80   transport.toggle

# ── feedback: OSC → MIDI ────────────────────────────────────────────────────
watch /live/song/get/is_playing      0  ->  cc 100
watch /live/view/get/selected_scene  0  ->  cc 101
watch /live/song/get/tempo           0  ->  cc 102  $v-50

# What MIDI cannot carry goes to state.json.
text  /live/song/get/track_names  ->  tracks

# ── phases: what settles itself, with nobody pressing anything ──────────────
# `boot` is what does NOT move during the show. It replays on its own as soon as
# Live answers after a silence — so on a Live restart, and when another set is
# loaded and puts the routing back wherever it likes. Make it idempotent.
on boot  /live/track/set/output_routing_type     -1 "Ext. Out"
on boot  /live/track/set/output_routing_channel  -1 "1/2"

# `song` is what PLAYING moves and what has to be straightened at the start of
# each piece. Fire it from whatever knows a new song was loaded.
on song  /live/song/set/metronome 0
on song  /live/view/set/selected_scene 0
trigger song  cc 51

# -1 is the Main (master) track: it is not in song.tracks, and a negative index
# would otherwise resolve silently to the LAST regular track. See patches/.
# Names with spaces go in double quotes — AbletonOSC matches an output routing by
# its display name, so "Ext. Out" is written exactly as Live shows it.
