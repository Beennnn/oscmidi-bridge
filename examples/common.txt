# Exemple minimal — à copier ailleurs et à adapter.
#
# La configuration réelle d'un rig ne vit PAS dans ce dépôt : elle décrit des
# numéros de CC, des index de pistes et des ports propres à une installation.
# Pointer la passerelle vers la sienne avec la variable OSCMIDI_CONFIG.

group   exemple
port    Ableton Loopback
channel 16
target  127.0.0.1:11000 -> 11001

# ── commandes : MIDI entrant → OSC ──────────────────────────────────────────
send  cc 20   /live/song/start_playing
send  cc 21   /live/song/stop_playing
send  cc 24   /live/song/tap_tempo
send  cc 25   $v+50   /live/song/set/tempo $a.0     # 0-127 → 50-177 bpm
send  cc 30   /live/view/set/selected_scene $a
send  cc 31   /live/scene/fire_selected

# ── gestes : calculés sur l'état observé ────────────────────────────────────
verb  cc 60   scene.prev
verb  cc 61   scene.next
verb  cc 80   transport.toggle

# ── retour d'état : OSC → MIDI ──────────────────────────────────────────────
watch /live/song/get/is_playing      0  ->  cc 100
watch /live/view/get/selected_scene  0  ->  cc 101
watch /live/song/get/tempo           0  ->  cc 102  $v-50

# Ce que le MIDI ne peut pas porter va dans state.json.
text  /live/song/get/track_names  ->  tracks
